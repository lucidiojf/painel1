// UPLOAD GITHUB (ABERTURA INSTANTÂNEA VIA RAW + GITHUB PAGES) async function enviarParaGitHub() { const token = document.getElementById('gh-token').value.trim() || localStorage.getItem('gh_user_token'); const inputFile = document.getElementById('gh-file'); const customName = document.getElementById('gh-custom-name').value.trim(); const statusTxt = document.getElementById('upload-status-text'); if (!token) return alert('Cole o Token do GitHub (ghp_...) e clique em Salvar!'); localStorage.setItem('gh_user_token', token); if (!inputFile.files || !inputFile.files.length) return alert('Selecione um arquivo!'); const file = inputFile.files[0]; statusTxt.style.color = '#0284c7'; statusTxt.innerText = 'Subindo p/ o GitHub... ⏳'; let nomeFinal = file.name; if (customName) { const ext = file.name.includes('.') ? file.name.split('.').pop() : ''; nomeFinal = customName.includes('.') ? customName : (ext ? `${customName}.${ext}` : customName); } // Remove espaços e acentos para garantir link perfeito nomeFinal = `${Date.now()}_${nomeFinal.normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[^a-zA-Z0-9._-]/g, "_")}`; try { const base64Content = await lerArquivoBase64(file); const urlApi = `https://api.github.com/repos/${GITHUB_USER}/${GITHUB_REPO}/contents/${nomeFinal}`; const resposta = await fetch(urlApi, { method: 'PUT', headers: { 'Authorization': `token ${token}`, 'Content-Type': 'application/json', 'Accept': 'application/vnd.github+json' }, body: JSON.stringify({ message: `Upload: ${nomeFinal}`, content: base64Content }) }); const resData = await resposta.json(); if (resposta.ok && resData.content) { // URL raw abre imediatamente sem esperar o delay do GitHub Pages const urlFinal = `https://raw.githubusercontent.com/${GITHUB_USER}/${GITHUB_REPO}/main/${nomeFinal}`; statusTxt.style.color = '#16a34a'; statusTxt.innerText = '✅ Enviado ao GitHub!'; inputFile.value = ''; document.getElementById('gh-custom-name').value = ''; if (abaAtual === 'notas') document.getElementById('nota-link').value = urlFinal; if (abaAtual === 'tarefas') document.getElementById('tarefa-link').value = urlFinal; if (abaAtual === 'despesas') document.getElementById('despesa-link').value = urlFinal; alert('Arquivo salvo com sucesso no GitHub!\nLink direto preenchido na nota.'); } else { throw new Error(resData.message || 'Erro de envio'); } } catch (e) { statusTxt.style.color = '#dc2626'; statusTxt.innerText = '❌ ' + e.message; alert('Erro no GitHub: ' + e.message); } } // DROPBOX SOLICITAÇÃO (SALVA O LINK REAL DA SUA CONTA) function abrirSolicitacaoDropbox() { const url = document.getElementById('db-req-url').value.trim(); if (!url || url === 'https://www.dropbox.com/request/' || !url.startsWith('http')) { return alert('Atenção: Cole o link completo da sua solicitação do Dropbox (ex: https://www.dropbox.com/request/SEU_CODIGO)!'); } localStorage.setItem('dropbox_req_url', url); window.open(url, '_blank'); }
